   200	def _repoint_python_package_imports(paths, all_nodes, all_edges, root) -> None:
   201	    """Repoint Python absolute-import edges to the real file node under a nested
   202	    (e.g. ``src/``) package root (#2072).
   203	
   204	    Absolute imports target an id derived from the dotted module path
   205	    (``_make_id('pkg.mod')`` -> ``pkg_mod``), but file-node ids are
   206	    scan-root-relative (``src_pkg_mod`` when the code lives under ``src/``), so
   207	    the edge dangles and is silently dropped — the graph loses most ``imports``
   208	    edges purely because of where the scan started. Build an alias map from the
   209	    dotted-module id to the real file-node id by detecting each ``.py`` file's
   210	    package root (the contiguous run of ancestor dirs carrying ``__init__.py``)
   211	    and rewrite matching ``imports``/``imports_from`` edge targets. Guards: never
   212	    shadow an existing node id, and drop an alias claimed by more than one file
   213	    (ambiguous -> leave dangling, as before). Files whose package root IS the
   214	    scan root are skipped (ids already coincide)."""
   215	    try:
   216	        root = Path(root).resolve()
   217	    except OSError:
   218	        root = Path(root)
   219	    node_ids = {n.get("id") for n in all_nodes if isinstance(n, dict)}
   220	    alias_to_files: dict[str, set[str]] = {}
   221	    for p in paths:
   222	        if p.suffix.lower() not in (".py", ".pyi"):
   223	            continue
   224	        try:
   225	            rel = Path(p).resolve().relative_to(root)
   226	        except (ValueError, OSError):
   227	            continue
   228	        parts = rel.parts
   229	        if len(parts) < 2:
   230	            continue  # top-level file: scan-root-relative id already matches
   231	        d = Path(p).resolve().parent
   232	        levels = 0
   233	        # Bounded by the number of dirs between the file and the scan root, so a
   234	        # pathological `/__init__.py` chain can't loop forever.
   235	        while levels < len(parts) - 1 and (d / "__init__.py").is_file():
   236	            levels += 1
   237	            d = d.parent
   238	        if levels == 0:
   239	            continue  # not inside a package (namespace pkg / loose module)
   240	        mod_parts = parts[-(levels + 1):]  # package dirs + the file itself
   241	        if len(mod_parts) == len(parts):
   242	            continue  # package root == scan root: file-node id already coincides
   243	        file_node = _file_node_id(rel)
   244	        alias = _make_id(str(Path(*mod_parts).with_suffix("")))
   245	        alias_to_files.setdefault(alias, set()).add(file_node)
   246	        if p.name in ("__init__.py", "__init__.pyi") and len(mod_parts) > 1:
   247	            # `import pkg` / `from pkg import x` targets the package-dir id.
   248	            pkg_alias = _make_id(str(Path(*mod_parts[:-1])))
   249	            alias_to_files.setdefault(pkg_alias, set()).add(file_node)
   250	    alias_map = {
   251	        a: next(iter(fs))
   252	        for a, fs in alias_to_files.items()
   253	        if len(fs) == 1 and a not in node_ids
   254	    }
   255	    if not alias_map:
   256	        return
   257	    for e in all_edges:
   258	        # Only repoint edges emitted from a Python file: a non-Python import edge
   259	        # (e.g. C# `using Pkg.Mod;`, Java/Go dotted imports) can have a dangling
   260	        # target string that coincides with a Python alias, and repointing it
   261	        # would fabricate a cross-language import edge (#2072 review).
   262	        if (
   263	            isinstance(e, dict)
   264	            and e.get("relation") in ("imports", "imports_from")
   265	            and str(e.get("source_file", "")).lower().endswith((".py", ".pyi"))
   266	        ):
   267	            tgt = e.get("target")
   268	            if tgt in alias_map:
   269	                e["target"] = alias_map[tgt]

   272	def _repoint_python_sibling_imports(paths, all_nodes, all_edges, root) -> None:
   273	    """Repoint Python sibling-import edges to the real file node in directories
   274	    without an __init__.py (#3430).
   275	
   276	    When a Python file below the scan root lives in a non-package directory
   277	    (no __init__.py in its immediate parent), plain imports of same-directory
   278	    modules (e.g. `scripts/main.py: import greeter`) target a bare name (`greeter`),
   279	    while the real file node is scan-root-relative (`scripts_greeter`). Because
   280	    the directory is not a package, `_repoint_python_package_imports` skips it
   281	    (levels == 0), leaving the edge dangling and causing downstream member calls
   282	    (`greeter.greet()`) to be dropped.
   283	
   284	    This pass is strictly importer-directory-local:
   285	    - Only applies when the importing file's parent directory has no __init__.py.
   286	    - Resolves only to unambiguous same-directory candidate modules in the scanned corpus.
   287	    - Never builds a global alias map and never searches outside the importer's directory.
   288	    - Preserves local aliases (e.g. `import greeter as g`).
   289	    """
   290	    try:
   291	        root = Path(root).resolve()
   292	    except OSError:
   293	        root = Path(root)
   294	
   295	    node_ids = {n.get("id") for n in all_nodes if isinstance(n, dict)}
   296	
   297	    # Index corpus modules by directory for non-package directories only.
   298	    # dir_path -> {module_name_id: set_of_file_node_ids}
   299	    dir_siblings: dict[Path, dict[str, set[str]]] = {}
   300	    for p in paths:
   301	        if p.suffix.lower() not in (".py", ".pyi"):
   302	            continue
   303	        try:
   304	            p_res = Path(p).resolve()
   305	            rel = p_res.relative_to(root)
   306	        except (ValueError, OSError):
   307	            continue
   308	
   309	        file_node = _file_node_id(rel)
   310	        if file_node not in node_ids:
   311	            continue
   312	
   313	        if p_res.name in ("__init__.py", "__init__.pyi"):
   314	            # Sibling package directory inside parent_dir (parent_dir / subpkg / __init__.py)
   315	            pkg_dir = p_res.parent
   316	            parent_dir = pkg_dir.parent
   317	            if not (parent_dir / "__init__.py").is_file() and not (parent_dir / "__init__.pyi").is_file():
   318	                mod_key = _make_id(pkg_dir.name)
   319	                dir_siblings.setdefault(parent_dir, {}).setdefault(mod_key, set()).add(file_node)
   320	            continue
   321	
   322	        d = p_res.parent
   323	        # PEP 328 guard: if the directory is a package, implicit relative imports
   324	        # are forbidden in Python 3. Do not index packages as loose sibling directories.
   325	        if (d / "__init__.py").is_file() or (d / "__init__.pyi").is_file():
   326	            continue
   327	
   328	        mod_key = _make_id(p_res.stem)
   329	        dir_siblings.setdefault(d, {}).setdefault(mod_key, set()).add(file_node)
   330	
   331	    # Require an unambiguous single candidate per module name in that directory.
   332	    dir_alias_map: dict[Path, dict[str, str]] = {
   333	        d: {
   334	            mod: next(iter(fns))
   335	            for mod, fns in mod_map.items()
   336	            if len(fns) == 1
   337	        }
   338	        for d, mod_map in dir_siblings.items()
   339	    }
   340	
   341	    if not dir_alias_map:
   342	        return
   343	
   344	    for e in all_edges:
   345	        if not (
   346	            isinstance(e, dict)
   347	            and e.get("relation") in ("imports", "imports_from")
   348	            and str(e.get("source_file", "")).lower().endswith((".py", ".pyi"))
   349	        ):
   350	            continue
   351	
   352	        sf = e.get("source_file")
   353	        if not sf:
   354	            continue
   355	        try:
   356	            sf_path = Path(sf)
   357	            if not sf_path.is_absolute():
   358	                sf_path = (root / sf_path).resolve()
   359	            else:
   360	                sf_path = sf_path.resolve()
   361	        except OSError:
   362	            continue
   363	
   364	        imp_dir = sf_path.parent
