    64	def _alive(pid: int) -> bool:
    65	    if os.name == "nt":
    66	        import ctypes
    67	
    68	        k32 = ctypes.windll.kernel32
    69	        h = k32.OpenProcess(0x1000, False, pid)  # PROCESS_QUERY_LIMITED_INFORMATION
    70	        if not h:
    71	            return False
    72	        code = ctypes.c_ulong()
    73	        k32.GetExitCodeProcess(h, ctypes.byref(code))
    74	        k32.CloseHandle(h)
    75	        return code.value == 259  # STILL_ACTIVE
    76	    try:  # pragma: no cover - POSIX
    77	        os.kill(pid, 0)
    78	    except ProcessLookupError:
    79	        return False
    80	    except PermissionError:
    81	        return True
    82	    return True
