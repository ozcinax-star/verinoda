   181	    """Formalize a correction of an exclusive claim ("only F calls P") and check it.
   182	
   183	    The corrected allowed set = files the correction text names that exist.
   184	    The statement is statically verified only if the pattern occurs in every
   185	    named file and nowhere else (same scan as critique's exclusivity check).
   186	    """
   187	    spec = claim.get("spec") or {}
   188	    rx = re.compile(spec["pattern"])
   189	    named = []
   190	    for tok in re.findall(r"[\w./\\-]+\.[A-Za-z0-9]+", correction):
   191	        rel = tok.replace("\\", "/")
   192	        rel = rel[2:] if rel.startswith("./") else rel
   193	        if (repo / rel).is_file() and rel not in named:
   194	            named.append(rel)
   195	    if not named:
   196	        return {"formalized": False, "why": "the correction names no existing file; exclusivity cannot be re-checked"}
   197	    hits, _ = _scan(repo, rx, None, suffixes=set(_EXCLUSIVE_SUFFIXES))
   198	    inside = [h for h in hits if h[0] in named]
   199	    outside = [h for h in hits if h[0] not in named]
   200	    missing = [f for f in named if not any(h[0] == f for h in inside)]
   201	    evs: list[tuple[str, str]] = []
   202	    for f in named:
   203	        for rel, i, _ in [h for h in inside if h[0] == f][:2]:
   204	            eid = rs.record_evidence(store, evmod.source_evidence(repo, rel, i, commit=commit,
   205	                                                                  meta={"check": "exclusivity (correction)"}))
   206	            if eid:
   207	                evs.append((eid, "supports"))
   208	    unc = []
   209	    if outside:
   210	        unc.append("pattern also occurs outside the named files: "
   211	                   + ", ".join(f"{r}:{i}" for r, i, _ in outside[:5]))
   212	    if missing:
   213	        unc.append("named file(s) do not match the pattern: " + ", ".join(missing))
   214	    return {"formalized": True, "allowed_files": named, "evidence": evs,
   215	            "status": "statically_verified" if not outside and not missing and evs else "weak_inference",
   216	            "uncertainties": unc, "outside": [f"{r}:{i}" for r, i, _ in outside[:5]], "missing": missing}
