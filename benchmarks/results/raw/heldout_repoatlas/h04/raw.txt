   128	def _map_like(d: dict) -> bool:
   129	    """A dict used as a mapping (name -> data), not a record with fixed fields.
   130	
   131	    Heuristic: at least 4 entries whose values are all dicts, all lists or all
   132	    numbers (e.g. ``env_vars``, ``affected_files``, ``churn``). Records - hops,
   133	    evidence rows, claims - mix value types or hold strings and are never cut
   134	    field by field.

   136	    if len(d) < 4:
   137	        return False
   138	    kinds = set()
   139	    for v in d.values():
   140	        if isinstance(v, dict):
   141	            kinds.add("dict")
   142	        elif isinstance(v, list):
   143	            kinds.add("list")
   144	        elif isinstance(v, (int, float)) and not isinstance(v, bool):
   145	            kinds.add("num")
   146	        else:
   147	            return False
   148	    return len(kinds) == 1

   179	def cap_response(obj: dict, limit: int = MAX_RESPONSE_CHARS, *, first: tuple[str, ...] = (),
   180	                 keep_tail: tuple[str, ...] = ("history",)) -> dict:
   181	    """Cut ``obj`` until its compact JSON (truncation note included) fits ``limit`` chars.
   182	
   183	    Top-level lists named in ``first`` are shortened before anything else.
   184	    Then the largest list (or name->data map) anywhere in the tree is cut, or
   185	    the longest string when it dominates. Cuts are proportional to the excess,
   186	    so the result stays close to the limit. Lists under a key in ``keep_tail``
   187	    keep their last items (newest history) instead of the first. Nothing is
   188	    re-ordered or invented; ``truncation.cut`` says what was cut (kept/total).
   189	    Top-level ``coverage`` (method and limits of a heuristic view) is never cut.
   190	    """
   191	    obj = _jsonable(obj)
   192	    if _size(obj) <= limit:
   193	        return obj
   194	    notes: dict[str, dict] = {}
   195	
   196	    def note() -> dict:
   197	        cut = dict(list(notes.items())[:25])
   198	        if len(notes) > 25:
   199	            cut["..."] = {"more_cut_fields": len(notes) - 25}
   200	        return {"max_chars": limit, "cut": cut,
   201	                "hint": "narrow the request (smaller max_items, explicit targets, one view) "
   202	                        "or run the matching `repoatlas ... --json` CLI command for the full result"}
   203	
   204	    def excess() -> int:
   205	        # size of obj + ',"truncated":true,"truncation":<note>' minus the limit
   206	        return _size(obj) + 32 + _size(note()) - limit
   207	
   208	    def record(path: tuple, total: int, kept: int) -> None:
   209	        k = _path_str(path)
   210	        prev = notes.get(k)
   211	        notes[k] = {"kept": kept, "total": prev["total"] if prev else total}
   212	
   213	    def cut_seq(path: tuple, node: list | dict, size: int, over: int) -> None:
   214	        n = len(node)
   215	        keep = int(n * (size - over) / size) if size > over else 1
   216	        keep = max(1, min(n - 1, keep))
   217	        record(path, n, keep)
   218	        if isinstance(node, dict):
   219	            for k in list(node)[keep:]:
   220	                del node[k]
   221	        elif bool(path) and path[-1] in keep_tail:
   222	            del node[: n - keep]
   223	        else:
   224	            del node[keep:]
   225	
   226	    for key in first:
   227	        while isinstance(obj.get(key), list) and len(obj[key]) > 1 and (over := excess()) > 0:
   228	            cut_seq((key,), obj[key], _size(obj[key]), over)
   229	
   230	    for _ in range(400):
   231	        over = excess()
   232	        if over <= 0:
   233	            break
   234	        best: dict = {"c": (0, None, None, (), None), "s": (0, None, None, (), None)}
   235	        _scan(obj, None, None, (), best)
   236	        c_size, _, _, c_path, c_node = best["c"]
   237	        s_size, s_parent, s_key, s_path, s_node = best["s"]
   238	        if s_node is not None and (c_node is None or s_size * 2 >= c_size):
   239	            new_len = max(200, min(len(s_node) - 1, len(s_node) - over - 16))
   240	            s_parent[s_key] = s_node[:new_len] + "...[truncated]"
   241	            record(s_path, len(s_node), new_len)
   242	        elif c_node is not None:
   243	            cut_seq(c_path, c_node, c_size, over)
   244	        else:
   245	            break  # nothing left that can be cut safely
   246	    unmet = excess() > 0
   247	    obj["truncated"] = True
   248	    obj["truncation"] = note()
   249	    if unmet:  # only protected/indivisible content is left; say so instead of cutting it
   250	        obj["truncation"]["over_limit"] = True
   251	    return obj
