   218	                "primary_source_verified": "primary_source_verified requires a supporting primary source",
   219	                "observed": "observed requires verifying evidence"}[status]
   220	        return f"{base} and states the claim; best group: {sorted(problems, key=len)[0]}"

   248	def check_status(status: str, evs: list[dict], source_checks: dict[str, bool] | None = None, *,
   249	                 claim: dict | None = None, repo: Path | None = None, grades: dict[str, str] | None = None
   250	                 ) -> str | None:
   251	    """Return None when ``status`` is allowed for this evidence, else the reason.
   252	
   253	    ``claim`` (with ``repo``) grades the evidence for relevance (or pass
   254	    precomputed ``grades``). Without either, only the evidence types, groups,
   255	    liveness and refutations are checked - a pure query about evidence. Every
   256	    status that reaches the store goes through :meth:`Claims.set_status`,
   257	    which always passes the claim, so irrelevant evidence cannot verify a
   258	    stored claim on any path.
   259	    """
   260	    if grades is None and claim is not None:
   261	        grades = grade_evidence(claim, evs, repo)
   262	    ctx = _Ctx(evs, source_checks, grades, (claim or {}).get("kind"))
   263	    why = _allows(status, ctx)
   264	    if why is not None:
   265	        return why
   266	    if ctx.ref_heur and status in ORDER:
   267	        lowered = step_down(_top(ctx))
   268	        if ORDER.index(status) < ORDER.index(lowered):
   269	            return f"a heuristic refutation lowers the claim one step (to {lowered})"
   270	    return None

   273	def best_status(evs: list[dict], ceiling: str | None = None,
   274	                source_checks: dict[str, bool] | None = None, *, claim: dict | None = None,
   275	                repo: Path | None = None, grades: dict[str, str] | None = None) -> tuple[str, list[str]]:
   276	    """Strongest status the evidence allows, not above ``ceiling``.
   277	
   278	    Definitive refutation with no live support at all gives ``contradicted``.
   279	    """
   280	    if grades is None and claim is not None:
   281	        grades = grade_evidence(claim, evs, repo)
   282	    ctx = _Ctx(evs, source_checks, grades, (claim or {}).get("kind"))
   283	    if ctx.ref_def and not ctx.sup:
   284	        return "contradicted", ["only refuting evidence is recorded"]
   285	    reasons: list[str] = []
   286	    start = ORDER.index(ceiling) if ceiling in ORDER else 0
   287	    for st in ORDER[start:]:

   289	        if why is None:
   290	            return st, reasons
   291	        reasons.append(why)
