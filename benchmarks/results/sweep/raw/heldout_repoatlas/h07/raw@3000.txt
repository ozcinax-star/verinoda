    96	    def _check(self) -> None:
    97	        if self.exhausted_reason is not None:
    98	            return

   100	            self.exhausted_reason = f"time budget {self.seconds}s spent"
   101	        elif self.calls > self.tool_calls:
   102	            self.exhausted_reason = f"tool-call budget {self.tool_calls} spent"
