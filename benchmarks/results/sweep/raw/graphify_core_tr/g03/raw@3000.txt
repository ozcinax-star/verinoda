    82

   102	def mint_external_stubs_in_data(data: dict) -> None:
   103	    """Mint external stub nodes for import-family links with an undeclared target.
   104	
   105	    The ``--no-cluster`` / incremental write path serializes the raw merged
   106	    extraction directly rather than going through :func:`build_from_json`, so it
   107	    needs the same stubbing to keep the on-disk graph.json free of undeclared
   108	    edge endpoints (#2873). Idempotent: a graph already carrying its stubs (from
   109	    a prior run or the clustered path) is left unchanged.
   110	    """
   111	    nodes = data.get("nodes")
   112	    links = data.get("links")
   113	    if not isinstance(links, list):
   114	        links = data.get("edges")
   115	    if not isinstance(nodes, list) or not isinstance(links, list):
   116	        return
   117	    declared = {n.get("id") for n in nodes if isinstance(n, dict)}
   118	    minted: set[str] = set()
   119	    for e in links:
   120	        if not isinstance(e, dict) or e.get("relation") not in _EXTERNAL_STUB_RELATIONS:
   121	            continue
   122	        src, tgt = e.get("source"), e.get("target")
   123	        if (
   124	            src in declared
   125	            and isinstance(tgt, str)
   126	            and tgt not in declared
   127	            and tgt not in minted
   128	        ):

   130	                "id": tgt,
   131	                "label": tgt,
   132	                "file_type": "concept",
   133	                "type": "external",
   134	                "external": True,
   135	                "source_file": "",
   136	            })
   137	            minted.add(tgt)
