    70	        except Exception:

   104	def _max_server_contexts() -> int:
   105	    """Return the project-context LRU capacity (default 8, minimum 1).
   106	
   107	    ``GRAPHIFY_MAX_CONTEXTS`` overrides the default. Invalid or blank values
   108	    use 8; zero and negative values clamp to 1, since each request needs a
   109	    graph context. The server's configured default graph is pinned separately

   112	    raw = os.environ.get("GRAPHIFY_MAX_CONTEXTS", "").strip()

   121	class _GraphContextCache:
   122	    """Thread-safe graph contexts: one pinned default plus an LRU of projects."""
   123	
   124	    def __init__(self, max_contexts: int):
   125	        self._max_contexts = max_contexts
   126	        self._entries: OrderedDict[str, dict] = OrderedDict()
   127	        self._pinned: dict[str, dict] = {}
   128	        self._lock = threading.Lock()
   129	
   130	    def _load_entry(self, resolved_path: str, key: tuple[int, int]) -> dict:
   131	        """Build one entry for an already-resolved path and known file key.
   132	
   133	        ``_load_graph`` is also used by the CLI, where invalid input terminates
   134	        the process. A client-supplied ``project_path`` must instead become a
   135	        tool error, so the shared MCP server can continue serving other graphs.
   136	        """
   137	        try:
   138	            graph = _load_graph(resolved_path)
   139	        except SystemExit as exc:
   140	            raise RuntimeError(f"could not load graph.json at {resolved_path}") from exc
   141	        # Warm the index before exposing the graph so its first query does not
   142	        # pay the expensive build cost.
   143	        _get_trigram_index(graph)
   144	        communities = _communities_from_graph(graph)
   145	        entry = {
   146	            "key": key,
   147	            "G": graph,
   148	            "communities": communities,
   149	        }
   150	        return entry
   151	
   152	    def load(self, resolved_path: str, *, pinned: bool = False) -> tuple[nx.Graph, dict[int, list[str]]]:
   153	        """Return a fresh context, retaining project contexts by LRU order.
   154	
   155	        ``resolved_path`` is resolved by the caller, making this method the
   156	        sole owner of file statting and cache-key construction.
   157	
   158	        ``pinned=True`` is reserved for the server's configured default graph;
   159	        it remains warm without consuming a project-cache slot.
   160	        """
   161	        with self._lock:
   162	            try:
   163	                stat_result = Path(resolved_path).stat()
   164	            except FileNotFoundError:
   165	                raise FileNotFoundError(f"graph.json not found: {resolved_path}") from None
   166	            key = (stat_result.st_mtime_ns, stat_result.st_size)
   167	            entries = self._pinned if pinned else self._entries
   168	            entry = entries.get(resolved_path)
   169	            if entry is not None and entry["key"] == key:
   170	                if not pinned:
   171	                    self._entries.move_to_end(resolved_path)
   172	                return entry["G"], entry["communities"]
   173	
   174	            entry = self._load_entry(resolved_path, key)
   175	            entries[resolved_path] = entry
   176	            if not pinned:
   177	                self._entries.move_to_end(resolved_path)
   178	                while len(self._entries) > self._max_contexts:
   179	                    self._entries.popitem(last=False)
   180	            return entry["G"], entry["communities"]

   191	def _search_tokens(text: str) -> list[str]:
   192	    """Split text into word tokens, stripping punctuation and diacritics.
   193	
   194	    `_` is a separator, exactly like `-`. `\\w` counts underscore as a word
   195	    character but not hyphen, so `graph_first_guard` stayed one token while the
   196	    label `graph-first-guard.py` split into three — and the query matched
   197	    nothing. Both the query and the node label pass through here, so splitting
   198	    on `_` keeps the two sides consistent and snake_case lookups still resolve
   199	    (their tokens simply match the same way). Found 2026-07-29: the graph could
   200	    not find the underscore spelling of its own `local_id`.
   201	    """
   202	    return re.findall(r"[^\W_]+", _strip_diacritics(str(text)).lower())
