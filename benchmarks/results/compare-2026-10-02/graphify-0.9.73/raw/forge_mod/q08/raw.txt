     1	# Emberforge (example mod)
     2	
     3	A small, fictional NeoForge 1.21.1 mod used as a test corpus for Verinoda. It builds on
     4	paper only: textures (`.png`) are left out so the example stays plain text, and nothing
     5	here has been run in a game.
     6	
     7	Adds:
     8	
     9	- **Ember Forge** - a block that burns fuel into heat and slowly forges its input with
    10	  `emberforge:forging` recipes. Right-click opens it; a Forge Hammer stokes it.
    11	- **Cinder Ore** - generated in a few overworld biomes; forged into two Cinder Ingots
    12	  (a furnace gives one).
    13	- `/emberforge reset` and `/emberforge heat <pos>` for operators.
    14	
    15	Layout:
    16	
    17	- `src/main/java` - Java code (registries, block, block entity, menu, recipe, network, events)
    18	- `src/main/kotlin` - Kotlin code (heat rules, forge screen, commands); needs Kotlin for Forge
    19	- `src/main/resources/data/emberforge` - recipes, tags, loot tables, worldgen, advancement, functions
    20	- `src/main/resources/assets/emberforge` - blockstates, models, `en_us` / `tr_tr` translations
    21	- `config/emberforge-common.toml` - the default common config, for reference
    22	- `src/test/java` - JUnit tests

    75	    public static void serverTick(Level level, BlockPos pos, BlockState state, EmberForgeBlockEntity forge) {
    76	        boolean changed = false;
    77	
    78	        ItemStack fuel = forge.items.getStackInSlot(SLOT_FUEL);
    79	        int fuelHeat = HeatMath.heatForFuel(fuel);
    80	        if (fuelHeat > 0 && forge.heat + fuelHeat <= EmberConfig.MAX_HEAT.get()) {
    81	            forge.items.extractItem(SLOT_FUEL, 1, false);
    82	            forge.heat = HeatMath.addHeat(forge.heat, fuelHeat);
    83	            changed = true;
    84	        }
    85	
    86	        if (level.getGameTime() % 40 == 0 && level.getBlockState(pos.below()).is(ModTags.FORGE_HEAT_SOURCES)) {
    87	            forge.heat = HeatMath.addHeat(forge.heat, 2);
    88	            changed = true;
    89	        }
    90	
    91	        ItemStack input = forge.items.getStackInSlot(SLOT_INPUT);
    92	        Optional<RecipeHolder<ForgingRecipe>> match = level.getRecipeManager()
    93	                .getRecipeFor(ModRecipes.FORGING_TYPE.get(), new SingleRecipeInput(input), level);
    94	        if (match.isPresent() && forge.canOutput(match.get().value().result())) {
    95	            ForgingRecipe recipe = match.get().value();
    96	            forge.forgeTime = recipe.forgeTime();
    97	            forge.progress += HeatMath.progressPerTick(forge.heat, recipe.minHeat());
    98	            if (forge.progress >= recipe.forgeTime()) {
    99	                forge.finish(recipe);
   100	            }
   101	            changed = true;
   102	        } else if (forge.progress != 0) {
   103	            forge.progress = 0;
   104	            changed = true;
   105	        }
   106	
   107	        int cooled = cooled(forge.heat, level.getGameTime(), EmberConfig.DECAY_INTERVAL.get());
   108	        if (cooled != forge.heat) {
   109	            forge.heat = cooled;
   110	            changed = true;
   111	        }
   112	
   113	        boolean lit = forge.heat > 0;
   114	        if (state.getValue(EmberForgeBlock.LIT) != lit) {
   115	            level.setBlock(pos, state.setValue(EmberForgeBlock.LIT, lit), 3);
   116	        }
   117	        if (changed) {
   118	            setChanged(level, pos, state);
   119	        }
   120	    }
