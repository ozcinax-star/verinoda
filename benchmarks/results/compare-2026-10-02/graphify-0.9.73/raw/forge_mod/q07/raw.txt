    10	import net.minecraft.core.BlockPos
    11	import net.minecraft.network.chat.Component
    12	
    13	/** /emberforge reset and /emberforge heat <pos>. */
    14	object ForgeCommands {
    15	    @JvmStatic
    16	    fun register(dispatcher: CommandDispatcher<CommandSourceStack>) {
    17	        dispatcher.register(
    18	            Commands.literal("emberforge")
    19	                .then(
    20	                    Commands.literal("reset")
    21	                        .requires { it.hasPermission(2) && EmberConfig.ALLOW_RESET_COMMAND.get() }
    22	                        .executes { ctx -> if (ForgeFunctions.resetForges(ctx.source)) 1 else 0 }
    23	                )
    24	                .then(
    25	                    Commands.literal("heat")
    26	                        .then(
    27	                            Commands.argument("pos", BlockPosArgument.blockPos())
    28	                                .executes { ctx -> reportHeat(ctx.source, BlockPosArgument.getLoadedBlockPos(ctx, "pos")) }
    29	                        )
    30	                )
    31	        )
    32	    }
    33	
    34	    private fun reportHeat(source: CommandSourceStack, pos: BlockPos): Int {
    35	        val forge = source.level.getBlockEntity(pos) as? EmberForgeBlockEntity
    36	        if (forge == null) {
    37	            source.sendFailure(Component.translatable("command.emberforge.heat.not_a_forge"))
    38	            return 0
    39	        }
    40	        source.sendSuccess({ Component.translatable("command.emberforge.heat", forge.heat) }, false)
    41	        return forge.heat
    42	    }
    43	}

    11	
    12	    private ForgeFunctions() {
    13	    }
    14	
    15	    /** Runs the reset function at the source's position; false when the data pack does not provide it. */
    16	    public static boolean resetForges(CommandSourceStack source) {
    17	        MinecraftServer server = source.getServer();
    18	        ResourceLocation id = ResourceLocation.parse(RESET_FORGES);
    19	        if (server.getFunctions().get(id).isEmpty()) {
    20	            EmberForge.LOGGER.warn("Function {} is missing; is the mod's data pack disabled?", id);
    21	            return false;
    22	        }
    23	        server.getCommands().performPrefixedCommand(source, "function " + RESET_FORGES);
    24	        return true;
    25	    }
    26	}

==> README.md:1-22 <==
     1	# Emberforge (example mod)
     2	
     3	A small, fictional NeoForge 1.21.1 mod used as a test corpus for Verinoda. It builds on
     4	paper only: textures (`.png`) are left out so the example stays plain text, and nothing
     5	here has been run in a game.
     6	
     7	Adds:
     8	
     9	- **Ember Forge** - a block that burns fuel into heat and slowly forges its input with
    10	  `emberforge:forging` recipes. Right-click opens it; a Forge Hammer stokes it.
    11	- **Cinder Ore** - generated in a few overworld biomes; forged into two Cinder Ingots
    12	  (a furnace gives one).
    13	- `/emberforge reset` and `/emberforge heat <pos>` for operators.
    14	
    15	Layout:
    16	
    17	- `src/main/java` - Java code (registries, block, block entity, menu, recipe, network, events)
    18	- `src/main/kotlin` - Kotlin code (heat rules, forge screen, commands); needs Kotlin for Forge
    19	- `src/main/resources/data/emberforge` - recipes, tags, loot tables, worldgen, advancement, functions
    20	- `src/main/resources/assets/emberforge` - blockstates, models, `en_us` / `tr_tr` translations
    21	- `config/emberforge-common.toml` - the default common config, for reference
    22	- `src/test/java` - JUnit tests
