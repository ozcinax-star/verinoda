     8	/** Heat rules of the Ember Forge. */
     9	object HeatMath {
    10	    private const val HEAT_PER_EXTRA_STEP = 50
    11	    private const val MAX_STEPS = 4
    12	
    13	    /** Heat one fuel item adds; blaze rods count double. 0 for items that are not forge fuel. */
    14	    @JvmStatic
    15	    fun heatForFuel(stack: ItemStack): Int {
    16	        if (stack.isEmpty || !stack.`is`(ModTags.FORGE_FUELS)) return 0
    17	        val base = EmberConfig.HEAT_PER_FUEL.get()
    18	        return if (stack.`is`(Items.BLAZE_ROD)) base * 2 else base
    19	    }
    20	
    21	    @JvmStatic
    22	    fun addHeat(current: Int, amount: Int): Int = addHeat(current, amount, EmberConfig.MAX_HEAT.get())
    23	
    24	    @JvmStatic
    25	    fun addHeat(current: Int, amount: Int, maxHeat: Int): Int = (current + amount).coerceIn(0, maxHeat)
    26	
    27	    /**
    28	     * Forging progress per tick: nothing below the recipe's minimum heat, then 1,
    29	     * plus 1 for every full 50 heat above the minimum, at most 4.
    30	     */
    31	    @JvmStatic
    32	    fun progressPerTick(heat: Int, minHeat: Int): Int {
    33	        if (heat < minHeat) return 0
    34	        return (1 + (heat - minHeat) / HEAT_PER_EXTRA_STEP).coerceAtMost(MAX_STEPS)
    35	    }
