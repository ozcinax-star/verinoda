   314	    """Turkish letters, two Turkish function words, or one Turkish question word ("kayit nerede")."""
   315	    if tn.has_turkish(text):
   316	        return True
   317	    return any(w in tn.TR_QUESTION_WORDS for w in re.findall(r"[a-z]+", tn.fold_tr(text)))

   320	def _shadowed(word: str, lexicon) -> bool:
   321	    """Is a Turkish cue word also a domain word of this repository?"""
   322	    if lexicon is None:
   323	        return False
   324	    if lexicon.has(word):
   325	        return True
   326	    return any(hit["targets"] for hit in lexicon.seed([word]))
