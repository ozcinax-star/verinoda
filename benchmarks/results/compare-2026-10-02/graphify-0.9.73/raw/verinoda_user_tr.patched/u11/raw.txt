    96	@dataclass
    97	class Budget:
    98	    seconds: float = 60
    99	    tool_calls: int = 40
   100	    context_tokens: int = 6000
   101	    started: float = field(default_factory=time.monotonic)
   102	    calls: int = 0
   103	    chars: int = 0
   104	    exhausted_reason: str | None = None
   105	
   106	    def _check(self) -> None:
   107	        if self.exhausted_reason is not None:
   108	            return
   109	        if time.monotonic() - self.started > self.seconds:
   110	            self.exhausted_reason = f"time budget {self.seconds}s spent"
   111	        elif self.calls > self.tool_calls:
   112	            self.exhausted_reason = f"tool-call budget {self.tool_calls} spent"
   113	        elif self.chars // 4 > self.context_tokens:
   114	            self.exhausted_reason = f"context budget ~{self.context_tokens} tokens spent"
   115	
   116	    def spend(self, calls: int = 1, chars: int = 0) -> bool:
   117	        self.calls += calls
   118	        self.chars += chars
   119	        self._check()
   120	        return self.exhausted_reason is None
   121	
   122	    def fits(self, chars: int) -> bool:
   123	        """True when ``chars`` more returned context stay within budget (nothing is charged)."""
   124	        self._check()
   125	        if self.exhausted_reason is not None:
   126	            return False
   127	        if (self.chars + chars) // 4 > self.context_tokens:
   128	            self.exhausted_reason = f"context budget ~{self.context_tokens} tokens spent"
   129	            return False
   130	        return True
   131	
   132	    def work_ok(self) -> bool:
   133	        """Time and one more tool call are still available (context aside)."""
   134	        return time.monotonic() - self.started <= self.seconds and self.calls + 1 <= self.tool_calls
   135	
   136	    def remaining_seconds(self) -> float:
   137	        return self.seconds - (time.monotonic() - self.started)
   138	
   139	    @property
   140	    def ok(self) -> bool:
   141	        self._check()
   142	        return self.exhausted_reason is None
   143	
   144	    def usage(self) -> dict:
   145	        return {"elapsed_s": round(time.monotonic() - self.started, 3), "tool_calls": self.calls,
   146	                "context_chars": self.chars, "context_tokens_est": self.chars // 4,
   147	                "token_count_method": "chars/4 estimate", "exhausted": self.exhausted_reason}
