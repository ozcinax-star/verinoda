     1	# Glow Mod (example)
     2	
     3	A small, fictional Fabric mod for Minecraft 1.21.1, used as a public example and
     4	benchmark corpus for Verinoda's support of mod repositories (Java code plus data
     5	pack, assets and YAML configuration linked by resource ids). It has not been
     6	compiled or run in a game; textures (`.png`) are left out on purpose, so the
     7	item models point at files that do not exist here.
     8	
     9	What it adds: a wisp mob, three items (lantern staff, ember shard, wisp heart),
    10	a lantern ritual, the `/glow` operator command and one client key binding.
    11	
    12	Layout:
    13	
    14	- `src/main/java/com/example/glowmod/` - common (server + client) code
    15	- `src/client/java/` - client-only code
    16	- `src/gametest/java/` - Fabric game tests
    17	- `src/main/resources/data/glowmod/` - data pack: functions, recipe, loot table
    18	- `src/main/resources/assets/glowmod/` - item models and translations
    19	- `src/main/resources/glowmod.yml` - default settings, copied to `config/` on first start
    20	- `datapack/` - a standalone data pack for map makers (a copy of the wisp death function)
    21	- `reference/original-plugin/` - excerpts of the Paper plugin this mod was ported
    22	  from, kept for comparison; it is not built
    23	
    24	Some identifiers are Turkish (`etkinlestir` = enable, `baslat` = start,
    25	`kaydet` = record, `temizle` = clean up, `alaniAc` = clear the area), as in the
    26	private mod this example imitates.

     1	package com.example.glowplugin;
     2	
     3	import net.kyori.adventure.text.Component;
     4	import net.kyori.adventure.text.format.NamedTextColor;
     5	import org.bukkit.Location;
     6	import org.bukkit.Material;
     7	import org.bukkit.NamespacedKey;
     8	import org.bukkit.entity.Allay;
     9	import org.bukkit.entity.EntityType;
    10	import org.bukkit.inventory.ItemStack;
    11	import org.bukkit.persistence.PersistentDataType;
    12	import org.bukkit.plugin.Plugin;
    13	
    14	/** Paper has no custom entities, so a wisp is a named Allay with a marker key. */
    15	public final class Wisp {
    16	    private Wisp() {
    17	    }
    18	
    19	    public static Allay spawn(Plugin plugin, Location location) {
    20	        Allay allay = (Allay) location.getWorld().spawnEntity(location, EntityType.ALLAY);
    21	        allay.customName(Component.text("Wisp", NamedTextColor.AQUA));
    22	        allay.getPersistentDataContainer().set(new NamespacedKey(plugin, "wisp"), PersistentDataType.BYTE, (byte) 1);
    23	        return allay;
    24	    }
    25	
    26	    public static boolean isWisp(Plugin plugin, org.bukkit.entity.Entity entity) {
    27	        return entity.getPersistentDataContainer().has(new NamespacedKey(plugin, "wisp"), PersistentDataType.BYTE);
    28	    }
    29	
    30	    public static ItemStack heart() {
    31	        ItemStack stack = new ItemStack(Material.GLOWSTONE_DUST);
    32	        stack.editMeta(meta -> meta.displayName(Component.text("Wisp Heart", NamedTextColor.AQUA)));
    33	        return stack;
    34	    }
    35	}

     1	package com.example.glowplugin;
     2	
     3	import org.bukkit.Material;
     4	import org.bukkit.block.Block;
     5	import org.bukkit.event.EventHandler;
     6	import org.bukkit.event.Listener;
     7	import org.bukkit.event.block.Action;
     8	import org.bukkit.event.entity.EntityDeathEvent;
     9	import org.bukkit.event.player.PlayerInteractEvent;
    10	
    11	public final class WispListener implements Listener {
    12	    private final GlowPlugin plugin;
    13	
    14	    public WispListener(GlowPlugin plugin) {
    15	        this.plugin = plugin;
    16	    }
    17	
    18	    @EventHandler
    19	    public void onInteract(PlayerInteractEvent event) {
    20	        Block block = event.getClickedBlock();
    21	        if (event.getAction() != Action.RIGHT_CLICK_BLOCK || block == null || block.getType() != Material.SOUL_LANTERN) {
    22	            return;
    23	        }
    24	        if (event.getItem() != null && event.getItem().getType() == Material.BLAZE_ROD) {
    25	            Wisp.spawn(plugin, block.getLocation().add(0.5, 1, 0.5));
    26	        }
    27	    }
    28	
    29	    @EventHandler
    30	    public void onDeath(EntityDeathEvent event) {
    31	        if (Wisp.isWisp(plugin, event.getEntity())) {
    32	            event.getDrops().clear();
    33	            event.getDrops().add(Wisp.heart());
    34	        }
    35	    }
    36	}
