    14	/** Settings from config/glowmod.yml; every missing key falls back to {@link #DEFAULTS}. */
    15	public record GlowConfig(int ritualWispCount, double naturalChance, int blocksPerTick, int repairDelayTicks,
    16	                         int maxSummonDistance) {
    17	    private static final String FILE_NAME = "glowmod.yml";
    18	    private static final GlowConfig DEFAULTS = new GlowConfig(4, 0.02, 8, 40, 8);
    19	    private static volatile GlowConfig current = DEFAULTS;
    20	
    21	    public static GlowConfig get() {
    22	        return current;
    23	    }
    24	
    25	    /** Reads the file; on first start the commented default is copied out of the mod jar first. */
    26	    public static void load() {
    27	        Path file = FabricLoader.getInstance().getConfigDir().resolve(FILE_NAME);
    28	        try {
    29	            if (Files.notExists(file)) {
    30	                try (InputStream in = GlowConfig.class.getResourceAsStream("/" + FILE_NAME)) {
    31	                    Files.copy(Objects.requireNonNull(in, "default " + FILE_NAME + " missing from the jar"), file);
    32	                }
    33	            }
    34	            try (Reader reader = Files.newBufferedReader(file)) {
    35	                Map<String, Object> root = new Yaml().load(reader);
    36	                current = parse(root == null ? Map.of() : root);
    37	            }
    38	        } catch (IOException | RuntimeException e) {
    39	            GlowMod.LOGGER.error("could not read {}, using defaults", file, e);
    40	            current = DEFAULTS;
    41	        }
    42	    }
    43	
    44	    private static GlowConfig parse(Map<String, Object> root) {
    45	        Map<String, Object> wisp = section(root, "wisp");
    46	        Map<String, Object> repair = section(root, "repair");
    47	        Map<String, Object> network = section(root, "network");
    48	        return new GlowConfig(
    49	                number(wisp, "ritual_count", DEFAULTS.ritualWispCount()).intValue(),
    50	                number(wisp, "natural_chance", DEFAULTS.naturalChance()).doubleValue(),
    51	                number(repair, "blocks_per_tick", DEFAULTS.blocksPerTick()).intValue(),
    52	                number(repair, "delay_ticks", DEFAULTS.repairDelayTicks()).intValue(),
    53	                number(network, "max_summon_distance", DEFAULTS.maxSummonDistance()).intValue());
    54	    }
    55	
    56	    @SuppressWarnings("unchecked")
    57	    private static Map<String, Object> section(Map<String, Object> root, String key) {
    58	        return root.get(key) instanceof Map<?, ?> map ? (Map<String, Object>) map : Map.of();
    59	    }

     1	# Glow Mod (example)
     2	
     3	A small, fictional Fabric mod for Minecraft 1.21.1, used as a public example and
     4	benchmark corpus for Verinoda's support of mod repositories (Java code plus data
     5	pack, assets and YAML configuration linked by resource ids). It has not been
     6	compiled or run in a game; textures (`.png`) are left out on purpose, so the
     7	item models point at files that do not exist here.
     8	
     9	What it adds: a wisp mob, three items (lantern staff, ember shard, wisp heart),
    10	a lantern ritual, the `/glow` operator command and one client key binding.
    11	
    12	Layout:
    13	
    14	- `src/main/java/com/example/glowmod/` - common (server + client) code
    15	- `src/client/java/` - client-only code
    16	- `src/gametest/java/` - Fabric game tests
    17	- `src/main/resources/data/glowmod/` - data pack: functions, recipe, loot table
    18	- `src/main/resources/assets/glowmod/` - item models and translations
    19	- `src/main/resources/glowmod.yml` - default settings, copied to `config/` on first start
    20	- `datapack/` - a standalone data pack for map makers (a copy of the wisp death function)
    21	- `reference/original-plugin/` - excerpts of the Paper plugin this mod was ported
    22	  from, kept for comparison; it is not built
    23	
    24	Some identifiers are Turkish (`etkinlestir` = enable, `baslat` = start,
    25	`kaydet` = record, `temizle` = clean up, `alaniAc` = clear the area), as in the
    26	private mod this example imitates.

     2	
     3	import com.example.glowmod.config.GlowConfig;
     4	import com.example.glowmod.entity.Wisp;
     5	import com.example.glowmod.repair.RepairScheduler;
     6	import com.example.glowmod.util.Datapack;
     7	import net.minecraft.block.BlockState;
     8	import net.minecraft.block.Blocks;
     9	import net.minecraft.registry.tag.BlockTags;
    10	import net.minecraft.server.world.ServerWorld;
    11	import net.minecraft.util.math.BlockPos;
    12	import net.minecraft.util.math.Vec3d;
    13	
    14	/** The lantern ritual: the data pack plays the effects, the mod clears the circle and calls the wisps. */
    15	public final class Ritual {
    16	    private static final int RADIUS = 3;
    17	
    18	    private Ritual() {
    19	    }
    20	
    21	    public static void baslat(ServerWorld world, BlockPos altar) {
    22	        Datapack.runQualified(world, Vec3d.ofBottomCenter(altar), "glowmod:ritual/start");
    23	        alaniAc(world, altar);
    24	        int count = GlowConfig.get().ritualWispCount();
    25	        for (int i = 0; i < count; i++) {
    26	            double angle = 2 * Math.PI * i / count;
    27	            BlockPos at = altar.add((int) Math.round(Math.cos(angle) * RADIUS), 1,
    28	                    (int) Math.round(Math.sin(angle) * RADIUS));
    29	            Wisp.spawn(world, at);
    30	        }
    31	    }
    32	
    33	    /** Burns away grass and flowers around the altar; RepairScheduler puts them back later. */
    34	    private static void alaniAc(ServerWorld world, BlockPos altar) {
    35	        for (BlockPos pos : BlockPos.iterate(altar.add(-RADIUS, 1, -RADIUS), altar.add(RADIUS, 2, RADIUS))) {
    36	            BlockState state = world.getBlockState(pos);
    37	            if (state.isIn(BlockTags.REPLACEABLE_BY_TREES)) {
    38	                RepairScheduler.kaydet(world, pos.toImmutable(), state);
    39	                world.setBlockState(pos, Blocks.AIR.getDefaultState());
    40	            }
    41	        }
    42	    }
    43	}
