     1	# Glow Mod (example)
     2	
     3	A small, fictional Fabric mod for Minecraft 1.21.1, used as a public example and
     4	benchmark corpus for Verinoda's support of mod repositories (Java code plus data
     5	pack, assets and YAML configuration linked by resource ids). It has not been
     6	compiled or run in a game; textures (`.png`) are left out on purpose, so the
     7	item models point at files that do not exist here.
     8	
     9	What it adds: a wisp mob, three items (lantern staff, ember shard, wisp heart),
    10	a lantern ritual, the `/glow` operator command and one client key binding.
    11	
    12	Layout:
    13	
    14	- `src/main/java/com/example/glowmod/` - common (server + client) code
    15	- `src/client/java/` - client-only code
    16	- `src/gametest/java/` - Fabric game tests
    17	- `src/main/resources/data/glowmod/` - data pack: functions, recipe, loot table
    18	- `src/main/resources/assets/glowmod/` - item models and translations
    19	- `src/main/resources/glowmod.yml` - default settings, copied to `config/` on first start
    20	- `datapack/` - a standalone data pack for map makers (a copy of the wisp death function)
    21	- `reference/original-plugin/` - excerpts of the Paper plugin this mod was ported
    22	  from, kept for comparison; it is not built
    23	
    24	Some identifiers are Turkish (`etkinlestir` = enable, `baslat` = start,
    25	`kaydet` = record, `temizle` = clean up, `alaniAc` = clear the area), as in the
    26	private mod this example imitates.

     9	import net.minecraft.util.Identifier;
    10	import net.minecraft.util.Rarity;
    11	
    12	public final class ModItems {
    13	    public static final Item LANTERN_STAFF = register("lantern_staff", new Item(new Item.Settings().maxCount(1)));
    14	    public static final Item EMBER_SHARD = register("ember_shard", new Item(new Item.Settings()));
    15	    public static final Item WISP_HEART = register("wisp_heart", new Item(new Item.Settings().rarity(Rarity.UNCOMMON)));
    16	
    17	    private ModItems() {
    18	    }
    19	
    20	    private static Item register(String name, Item item) {
    21	        return Registry.register(Registries.ITEM, Identifier.of(GlowMod.MOD_ID, name), item);
    22	    }
    23	
    24	    /** Loads the class (so the fields above register) and puts the items into the creative tabs. */
    25	    public static void initialize() {
    26	        ItemGroupEvents.modifyEntriesEvent(ItemGroups.TOOLS).register(entries -> entries.add(LANTERN_STAFF));
    27	        ItemGroupEvents.modifyEntriesEvent(ItemGroups.INGREDIENTS).register(entries -> {
    28	            entries.add(EMBER_SHARD);
    29	            entries.add(WISP_HEART);
    30	        });
    31	    }
    32	}

    10	
    11	public final class ServerNetworking {
    12	    private ServerNetworking() {
    13	    }
    14	
    15	    public static void register() {
    16	        PayloadTypeRegistry.playC2S().register(SummonWispPayload.ID, SummonWispPayload.CODEC);
    17	        ServerPlayNetworking.registerGlobalReceiver(SummonWispPayload.ID, ServerNetworking::onSummonWisp);
    18	    }
    19	
    20	    private static void onSummonWisp(SummonWispPayload payload, ServerPlayNetworking.Context context) {
    21	        ServerPlayerEntity player = context.player();
    22	        if (!player.getMainHandStack().isOf(ModItems.LANTERN_STAFF)) {
    23	            return;
    24	        }
    25	        if (!player.getBlockPos().isWithinDistance(payload.pos(), GlowConfig.get().maxSummonDistance())) {
    26	            GlowMod.LOGGER.warn("{} asked for a wisp too far away at {}", player.getName().getString(), payload.pos());
    27	            return;
    28	        }
    29	        Wisp.spawn(player.getServerWorld(), payload.pos().up());
    30	    }
    31	}
