     1	# Glow Mod (example)
     2	
     3	A small, fictional Fabric mod for Minecraft 1.21.1, used as a public example and
     4	benchmark corpus for Verinoda's support of mod repositories (Java code plus data
     5	pack, assets and YAML configuration linked by resource ids). It has not been
     6	compiled or run in a game; textures (`.png`) are left out on purpose, so the
     7	item models point at files that do not exist here.
     8	
     9	What it adds: a wisp mob, three items (lantern staff, ember shard, wisp heart),
    10	a lantern ritual, the `/glow` operator command and one client key binding.
    11	
    12	Layout:
    13	
    14	- `src/main/java/com/example/glowmod/` - common (server + client) code
    15	- `src/client/java/` - client-only code
    16	- `src/gametest/java/` - Fabric game tests
    17	- `src/main/resources/data/glowmod/` - data pack: functions, recipe, loot table
    18	- `src/main/resources/assets/glowmod/` - item models and translations
    19	- `src/main/resources/glowmod.yml` - default settings, copied to `config/` on first start
    20	- `datapack/` - a standalone data pack for map makers (a copy of the wisp death function)
    21	- `reference/original-plugin/` - excerpts of the Paper plugin this mod was ported
    22	  from, kept for comparison; it is not built
    23	
    24	Some identifiers are Turkish (`etkinlestir` = enable, `baslat` = start,
    25	`kaydet` = record, `temizle` = clean up, `alaniAc` = clear the area), as in the
    26	private mod this example imitates.

     1	package com.example.glowmod.event;
     2	
     3	import com.example.glowmod.entity.Wisp;
     4	import com.example.glowmod.item.ModItems;
     5	import net.fabricmc.fabric.api.event.player.UseBlockCallback;
     6	import net.minecraft.block.Blocks;
     7	import net.minecraft.server.world.ServerWorld;
     8	import net.minecraft.util.ActionResult;
     9	import net.minecraft.util.math.BlockPos;
    10	
    11	public final class LanternEvents {
    12	    private static final int STAFF_COOLDOWN_TICKS = 100;
    13	
    14	    private LanternEvents() {
    15	    }
    16	
    17	    /** Lantern staff + right click on a soul lantern calls a wisp out of the lantern. */
    18	    public static void etkinlestir() {
    19	        UseBlockCallback.EVENT.register((player, world, hand, hit) -> {
    20	            if (world.isClient() || !player.getStackInHand(hand).isOf(ModItems.LANTERN_STAFF)) {
    21	                return ActionResult.PASS;
    22	            }
    23	            BlockPos pos = hit.getBlockPos();
    24	            if (!world.getBlockState(pos).isOf(Blocks.SOUL_LANTERN)) {
    25	                return ActionResult.PASS;
    26	            }
    27	            if (player.getItemCooldownManager().isCoolingDown(ModItems.LANTERN_STAFF)) {
    28	                return ActionResult.FAIL;
    29	            }
    30	            Wisp.spawn((ServerWorld) world, pos.up());
    31	            player.getItemCooldownManager().set(ModItems.LANTERN_STAFF, STAFF_COOLDOWN_TICKS);
    32	            return ActionResult.SUCCESS;
    33	        });
    34	    }
    35	}
