    14	public final class RepairScheduler {
    15	    private record Pending(ServerWorld world, BlockPos pos, BlockState state, long dueTick) {
    16	    }
    17	
    18	    private static final Deque<Pending> QUEUE = new ArrayDeque<>();
    19	
    20	    private RepairScheduler() {
    21	    }
    22	
    23	    public static void register() {
    24	        ServerTickEvents.END_SERVER_TICK.register(RepairScheduler::tick);
    25	        ServerLifecycleEvents.SERVER_STOPPING.register(RepairScheduler::temizle);
    26	    }

    28	    public static void kaydet(ServerWorld world, BlockPos pos, BlockState state) {
    29	        long due = world.getServer().getTicks() + GlowConfig.get().repairDelayTicks();
    30	        QUEUE.addLast(new Pending(world, pos, state, due));
    31	    }
    32	
    33	    private static void tick(MinecraftServer server) {
    34	        int budget = GlowConfig.get().blocksPerTick();
    35	        while (budget-- > 0 && !QUEUE.isEmpty() && QUEUE.peekFirst().dueTick() <= server.getTicks()) {
    36	            Pending p = QUEUE.pollFirst();
    37	            p.world().setBlockState(p.pos(), p.state());
    38	        }
    39	    }
    40	
    41	    /** The server is stopping: every block still waiting is put back at once, then the queue is emptied. */
    42	    private static void temizle(MinecraftServer server) {
    43	        for (Pending p : QUEUE) {
    44	            p.world().setBlockState(p.pos(), p.state());
    45	        }
