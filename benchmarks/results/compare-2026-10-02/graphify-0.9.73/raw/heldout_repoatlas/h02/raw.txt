   165	def path_escape(value: str) -> str | None:
   166	    """Why ``value`` names a location outside the repository copy, else None.
   167	
   168	    Purely lexical, so it is the same on every OS and needs no file system:
   169	    the copy never contains symlinks (files are copied by content), so a
   170	    path that stays inside lexically stays inside for real.
   171	    """
   172	    v = _strip_quotes(value).strip()
   173	    if not v:
   174	        return None
   175	    s = v.replace("\\", "/")
   176	    if s.startswith("~"):
   177	        return f"{value!r} is relative to the home directory"
   178	    if _DRIVE_RE.match(s) or s.startswith("/"):
   179	        return f"{value!r} is an absolute path"
   180	    if _URL_RE.match(s):
   181	        return f"{value!r} is a URL"
   182	    depth = 0
   183	    for part in s.split("/"):
   184	        if part == "..":
   185	            depth -= 1
   186	            if depth < 0:
   187	                return f"{value!r} leaves the repository copy through '..'"
   188	        elif part not in ("", "."):
   189	            depth += 1
   190	    return None

   215	def policy(argv: list[str], allowlist: list[str]) -> tuple[str, str | None]:
   216	    """``('allowlisted', None)`` or ``('risky', why)``.
   217	
   218	    Allowlisted means: argv starts with an allowlisted test runner, has no
   219	    shell metacharacters, no forbidden runner option, and no argument that
   220	    names a path outside the repository copy (see the module docstring).
   221	    """
   222	    if not argv:
   223	        return "risky", "empty command"
   224	    joined = " ".join(argv)
   225	    if SHELL_META.search(joined):
   226	        return "risky", "the command contains shell metacharacters"
   227	    raw = _arg0_path(argv[0]).name.lower().removesuffix(".exe")
   228	    matched = False
   229	    for exe in dict.fromkeys((raw, _exe_name(argv[0]))):
   230	        norm = " ".join([exe, *argv[1:]])
   231	        if any(norm == p or norm.startswith(p + " ") for p in allowlist):
   232	            matched = True
   233	            break
   234	    if not matched:
   235	        return "risky", f"{' '.join(argv[:3])!r} does not start with an allowlisted test runner"
   236	    forbidden = FORBIDDEN_OPTIONS.get(_runner(argv), {})
   237	    for tok in argv[1:]:
   238	        opt = _strip_quotes(tok).split("=", 1)[0]
   239	        if opt in forbidden:
   240	            return "risky", forbidden[opt]
   241	        for cand in _path_candidates(_strip_quotes(tok)):
   242	            why = path_escape(cand)
   243	            if why:
   244	                return "risky", f"argument {tok!r}: {why}; {ARGS_CONFINED}"
