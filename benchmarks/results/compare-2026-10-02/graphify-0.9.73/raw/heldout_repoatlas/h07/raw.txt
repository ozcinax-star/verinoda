    86	@dataclass
    87	class Budget:
    88	    seconds: float = 60
    89	    tool_calls: int = 40
    90	    context_tokens: int = 6000
    91	    started: float = field(default_factory=time.monotonic)
    92	    calls: int = 0
    93	    chars: int = 0
    94	    exhausted_reason: str | None = None
    95	
    96	    def _check(self) -> None:
    97	        if self.exhausted_reason is not None:
    98	            return
    99	        if time.monotonic() - self.started > self.seconds:
   100	            self.exhausted_reason = f"time budget {self.seconds}s spent"
   101	        elif self.calls > self.tool_calls:
   102	            self.exhausted_reason = f"tool-call budget {self.tool_calls} spent"
   103	        elif self.chars // 4 > self.context_tokens:
   104	            self.exhausted_reason = f"context budget ~{self.context_tokens} tokens spent"
   105	
   106	    def spend(self, calls: int = 1, chars: int = 0) -> bool:
   107	        self.calls += calls
   108	        self.chars += chars
   109	        self._check()
   110	        return self.exhausted_reason is None
   111	
   112	    def fits(self, chars: int) -> bool:
   113	        """True when ``chars`` more returned context stay within budget (nothing is charged)."""
   114	        self._check()
   115	        if self.exhausted_reason is not None:
   116	            return False
   117	        if (self.chars + chars) // 4 > self.context_tokens:
   118	            self.exhausted_reason = f"context budget ~{self.context_tokens} tokens spent"
   119	            return False
   120	        return True
   121	
   122	    def work_ok(self) -> bool:
   123	        """Time and one more tool call are still available (context aside)."""
   124	        return time.monotonic() - self.started <= self.seconds and self.calls + 1 <= self.tool_calls
   125	
   126	    def remaining_seconds(self) -> float:
   127	        return self.seconds - (time.monotonic() - self.started)
   128	
   129	    @property
   130	    def ok(self) -> bool:
   131	        self._check()
   132	        return self.exhausted_reason is None
   133	
   134	    def usage(self) -> dict:
   135	        return {"elapsed_s": round(time.monotonic() - self.started, 3), "tool_calls": self.calls,
   136	                "context_chars": self.chars, "context_tokens_est": self.chars // 4,
   137	                "token_count_method": "chars/4 estimate", "exhausted": self.exhausted_reason}
