   281	def _query_terms(question: str) -> list[str]:
   282	    """Split a query into searchable terms, segmenting Chinese text, then drop
   283	    question/filler words (`_QUERY_STOPWORDS`, English plus common German/
   284	    Romance-language fillers) so content words drive seeding. Falls back to the
   285	    unfiltered terms if the query is all stopwords, so a question like "how does
   286	    it work" or "wie funktioniert das" still seeds on something."""
   287	    terms: list[str] = []
   288	    for raw in question.split():
   289	        if _has_chinese(raw):
   290	            for seg in _segment_chinese(raw.lower().strip()):
   291	                seg = seg.strip()
   292	                if seg and _is_searchable(seg):
   293	                    terms.append(seg)
   294	        else:
   295	            # Strip punctuation without touching Unicode characters (avoid NFKD mangling non-Latin scripts)
   296	            for tok in re.findall(r"\w+", raw.lower()):
   297	                if _is_searchable(tok):
   298	                    terms.append(tok)
   299	    content = [t for t in terms if t not in _QUERY_STOPWORDS]
   300	    return content or terms

   346	def _compute_idf(G: nx.Graph, terms: list[str]) -> dict[str, float]:
   347	    """IDF weights for query terms, cached in G.graph['_idf_cache'].
   348	
   349	    Common terms like 'error' or 'exception' that match hundreds of nodes get
   350	    low weights; rare identifiers like 'FooBarService' get high weights.
   351	    Cache is stored on the graph object itself so it auto-invalidates when
   352	    a hot-reload replaces G with a new object.
   353	    """
   354	    cache: dict[str, float] = G.graph.setdefault("_idf_cache", {})
   355	    N = G.number_of_nodes() or 1
   356	    uncached = [t for t in terms if t not in cache]
   357	    if uncached:
   358	        df: dict[str, int] = {t: 0 for t in uncached}
   359	        for _, data in G.nodes(data=True):
   360	            norm_label = (
   361	                data.get("norm_label") or _strip_diacritics(data.get("label") or "")
   362	            ).lower()
   363	            for t in uncached:
   364	                if t in norm_label:
   365	                    df[t] += 1
   366	        for t in uncached:
   367	            cache[t] = math.log(1 + N / (1 + df[t]))
   368	    return {t: cache.get(t, math.log(1 + N)) for t in terms}
