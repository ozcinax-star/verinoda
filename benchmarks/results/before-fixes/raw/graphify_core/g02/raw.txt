     1	# assemble node+edge dicts into a NetworkX graph, preserving edge direction
     2	#
     3	# Node deduplication — three layers:
     4	#
     5	# 1. Within a file (AST): each extractor tracks a `seen_ids` set. A node ID is
     6	#    emitted at most once per file, so duplicate class/function definitions in
     7	#    the same source file are collapsed to the first occurrence.
     8	#
     9	# 2. Between files (build): NetworkX G.add_node() is idempotent — calling it
    10	#    twice with the same ID overwrites the attributes with the second call's
    11	#    values. Nodes are added in extraction order (AST first, then semantic),
    12	#    so if the same entity is extracted by both passes the semantic node
    13	#    silently overwrites the AST node. This is intentional: semantic nodes
    14	#    carry richer labels and cross-file context, while AST nodes have precise
    15	#    source_location. If you need to change the priority, reorder extractions
    16	#    passed to build().
    17	#
    18	# 3. Semantic merge (skill): before calling build(), the skill merges cached
    19	#    and new semantic results using an explicit `seen` set keyed on node["id"],
    20	#    so duplicates across cache hits and new extractions are resolved there
    21	#    before any graph construction happens.
    22	#

   102	def mint_external_stubs_in_data(data: dict) -> None:
   103	    """Mint external stub nodes for import-family links with an undeclared target.
   104	
   105	    The ``--no-cluster`` / incremental write path serializes the raw merged
   106	    extraction directly rather than going through :func:`build_from_json`, so it
   107	    needs the same stubbing to keep the on-disk graph.json free of undeclared
   108	    edge endpoints (#2873). Idempotent: a graph already carrying its stubs (from
   109	    a prior run or the clustered path) is left unchanged.
   110	    """
   111	    nodes = data.get("nodes")
   112	    links = data.get("links")
   113	    if not isinstance(links, list):
   114	        links = data.get("edges")
   115	    if not isinstance(nodes, list) or not isinstance(links, list):
   116	        return
   117	    declared = {n.get("id") for n in nodes if isinstance(n, dict)}
   118	    minted: set[str] = set()
   119	    for e in links:
   120	        if not isinstance(e, dict) or e.get("relation") not in _EXTERNAL_STUB_RELATIONS:
   121	            continue
   122	        src, tgt = e.get("source"), e.get("target")
   123	        if (
   124	            src in declared
   125	            and isinstance(tgt, str)
   126	            and tgt not in declared
   127	            and tgt not in minted
   128	        ):
   129	            nodes.append({
   130	                "id": tgt,
   131	                "label": tgt,
   132	                "file_type": "concept",
   133	                "type": "external",
   134	                "external": True,
   135	                "source_file": "",
   136	            })
   137	            minted.add(tgt)

   187	def _coerce_hyperedge_member_refs(he: dict, members: list) -> list:
   188	    """Coerce a hyperedge member list to hashable scalar ids, deduped in order.
   189	
   190	    LLM/subagent drift sometimes emits a member as an object (``{"id": "a_ts"}``)
   191	    instead of a bare id string. Left uncoerced, the dict member is unhashable,
   192	    so the semantic-rekey pass's ``_rekey.get(n, n)`` raised ``TypeError`` and
   193	    aborted the whole merge — destroying a completed extraction (#2486). Object
   194	    members collapse to their non-empty ``id`` (numeric ids str-coerced via
   195	    ``_coerce_id``, matching #2326); members with no usable id are dropped with
   196	    a stderr WARNING naming the hyperedge, never a crash. Hashable scalar refs
   197	    pass through unchanged. A hyperedge that loses every member this way falls
   198	    to the existing no-valid-members drop-with-warning in ``build_from_json``.
   199	    """
   200	    seen: set = set()
   201	    coerced: list = []
   202	    for ref in members:
   203	        if isinstance(ref, dict):
   204	            inner = _coerce_id(ref.get("id"))
   205	            if inner in (None, "") or not _hashable(inner):
   206	                print(
   207	                    f"[graphify] WARNING: hyperedge "
   208	                    f"'{he.get('id', '?')}' has a member object with no usable "
   209	                    f"'id' ({ref!r}); dropping that member.",
   210	                    file=sys.stderr,
   211	                )
   212	                continue
   213	            ref = inner
   214	        elif not _hashable(ref):
   215	            print(
   216	                f"[graphify] WARNING: hyperedge "
   217	                f"'{he.get('id', '?')}' has an unusable member reference "
   218	                f"{ref!r}; dropping that member.",
   219	                file=sys.stderr,
   220	            )
   221	            continue
   222	        if ref in seen:
   223	            continue
   224	        seen.add(ref)
   225	        coerced.append(ref)
   226	    return coerced

   229	def _normalize_hyperedge_members(he: object) -> None:
   230	    """Canonicalize a hyperedge's member list onto the `nodes` key, in place.
   231	
   232	    If `nodes` is already a list it wins (canonical), and only stray alias keys
   233	    are dropped. Otherwise the first alias (`members`, then `node_ids`) that is a
   234	    list is moved to `nodes`, with a single stderr WARNING naming the hyperedge
   235	    id and alias used. Leftover alias keys are always removed so downstream code
   236	    never re-reads them. Whichever branch supplied the list, member VALUES are
   237	    coerced to hashable scalar ids and deduped preserving order (#2486) — see
   238	    ``_coerce_hyperedge_member_refs``.
   239	    """
   240	    if not isinstance(he, dict):
   241	        return
   242	    if isinstance(he.get("nodes"), list):
   243	        he["nodes"] = _coerce_hyperedge_member_refs(he, he["nodes"])
   244	    else:
   245	        for alias in _HE_MEMBER_ALIASES:
   246	            val = he.get(alias)
   247	            if isinstance(val, list):
   248	                he["nodes"] = _coerce_hyperedge_member_refs(he, val)
   249	                print(
   250	                    f"[graphify] WARNING: hyperedge "
   251	                    f"'{he.get('id', '?')}' uses field '{alias}' instead of "
   252	                    f"'nodes'; normalizing.",
   253	                    file=sys.stderr,
   254	                )
   255	                break
   256	    # Drop any leftover alias keys regardless of which branch ran above.
   257	    for alias in _HE_MEMBER_ALIASES:
   258	        he.pop(alias, None)

   261	def _fold_node_aliases(node: dict) -> None:
   262	    """Fold legacy node field aliases onto canonical keys, in place (#2194).
   263	
   264	    ``name`` -> ``label`` and ``path`` -> ``source_file``. Uses an empty-check
   265	    (not mere key presence) so a node carrying ``label: ""``/``None`` next to a
   266	    real ``name`` is healed too. When the canonical field already holds a value
   267	    it wins and the alias key is left untouched. Without this fold an alias-only
   268	    node enters the graph with no label/source_file: it fails validation, gets
   269	    ``norm_label == ""`` (invisible to query/explain), and is excluded from every
   270	    label-keyed merge/dedup — a permanent ghost that ``graphify update``
   271	    re-feeds through build_from_json forever.
   272	    """
   273	    if not node.get("label") and isinstance(node.get("name"), str) and node["name"]:
   274	        node["label"] = node.pop("name")
   275	    if not node.get("source_file") and isinstance(node.get("path"), str) and node["path"]:
   276	        node["source_file"] = node.pop("path")

   279	def _fold_edge_aliases(edge: dict) -> None:
   280	    """Fold legacy edge field aliases onto canonical keys, in place (#2194).
   281	
   282	    ``type`` -> ``relation``. A ``confidence_score`` float with no ``confidence``
   283	    enum backfills ``confidence: "INFERRED"`` — never EXTRACTED (alias recovery
   284	    is not provenance) and never a threshold mapping of the float. The
   285	    ``confidence_score`` key itself is NOT popped: it is a legitimate companion
   286	    field that the edge loop sanitizes and to_json round-trips.
   287	
   288	    A NUMERIC ``confidence`` (pre-enum graphs stored the LLM pass's float —
   289	    1.0/0.95/0.9/0.85 — directly in the field) normalizes to ``INFERRED``:
   290	    numeric confidences only ever came from the LLM semantic pass, and
   291	    LLM-derived edges are INFERRED by definition. The original float moves to
   292	    ``confidence_score`` unless an explicit one is already present (the
   293	    companion field is the authority). Without this fold, every reload of a
   294	    pre-enum graph re-warns once per legacy edge, forever. ``bool`` is
   295	    excluded despite subclassing ``int``: ``True`` is not a score.
   296	    """
   297	    if not edge.get("relation") and isinstance(edge.get("type"), str) and edge["type"]:
   298	        edge["relation"] = edge.pop("type")
   299	    _conf = edge.get("confidence")
   300	    if isinstance(_conf, (int, float)) and not isinstance(_conf, bool):
   301	        if edge.get("confidence_score") is None:
   302	            edge["confidence_score"] = float(_conf)
   303	        edge["confidence"] = "INFERRED"
   304	    if not edge.get("confidence") and edge.get("confidence_score") is not None:
   305	        edge["confidence"] = "INFERRED"

   308	def _coerce_id(value: object) -> object:
   309	    """Return a str for a numeric id, else the value unchanged.
   310	
   311	    ``bool`` is excluded despite subclassing ``int``: an id of ``True`` is not a
   312	    number the model meant to name a node, and ``"True"`` would invent a label.
   313	    """
   314	    if isinstance(value, bool) or not isinstance(value, (int, float)):
   315	        return value
   316	    return str(value)

   319	def _hashable(value: object) -> bool:
   320	    """True when value can be a dict key / set member (same probe as the
   321	    inline ``try: hash(m)`` in build_from_json's hyperedge revalidation)."""
   322	    try:
   323	        hash(value)
   324	    except TypeError:
   325	        return False
   326	    return True

   329	def _coerce_non_string_ids(extraction: dict) -> None:
   330	    """Coerce numeric node ids and edge/hyperedge references to str, in place (#2326).
   331	
   332	    A backend can emit ``{"id": 10}`` where the schema says ``{"id": "10"}``.
   333	    Every id consumer downstream assumes ``str``, so one int id aborted the build
   334	    in three places: ``_pick_winner``'s ``_CHUNK_SUFFIX.search(n["id"])`` raised
   335	    ``TypeError: expected string or bytes-like object``, and ``build_from_json``'s
   336	    ``sorted(node_set)`` raised ``'<' not supported between instances of 'str'
   337	    and 'int'`` — the latter for a lone node with nothing to dedup at all.
   338	    Coercing keeps the node and its edges rather than dropping either, which is
   339	    the same tolerate-and-heal treatment loose backend output already gets at the
   340	    parse chokepoint (#1631) and in the alias folds (#2194).
   341	
   342	    Endpoints and hyperedge members are coerced with the nodes, not after: a
   343	    node-only coercion would renumber ``10`` to ``"10"`` and leave every edge
   344	    pointing at the vanished ``10``, trading a loud crash for a silently
   345	    disconnected graph. The legacy ``from``/``to`` endpoint aliases are included
   346	    because dedup reads them directly (#803).
   347	
   348	    Runs in BOTH ``build`` (before dedup, which keys on id) and
   349	    ``build_from_json`` (the direct entry that reloads a persisted graph), for
   350	    the same two-site reason as the ``_fold_node_aliases`` fold (#2194). It is
   351	    idempotent, so the nested call on the ``build`` path is a no-op.
   352	
   353	    Non-numeric non-str ids (``None``, lists, dicts) are left alone for
   354	    ``validate_extraction`` to report: ``str(None) == "None"`` would fabricate a
   355	    node id that no edge references.
   356	    """
   357	    for node in extraction.get("nodes") or ():
   358	        if isinstance(node, dict) and "id" in node:
   359	            node["id"] = _coerce_id(node["id"])
   360	    for edge in extraction.get("edges") or ():
   361	        if not isinstance(edge, dict):
   362	            continue
   363	        for key in ("source", "target", "from", "to"):
   364	            if key in edge:
   365	                edge[key] = _coerce_id(edge[key])
   366	    for he in extraction.get("hyperedges") or ():
   367	        if not isinstance(he, dict):
   368	            continue
   369	        members = he.get("nodes")
   370	        if isinstance(members, list):
   371	            he["nodes"] = [_coerce_id(ref) for ref in members]

   374	def _norm_source_file(p: str | None, root: str | None = None) -> str | None:
   375	    """Normalize path separators and relativize absolute paths.
   376	
   377	    Converts backslashes to forward slashes (Windows compatibility) and, when
   378	    root is provided, strips the absolute prefix from paths produced by semantic
   379	    subagents so source_file is always repo-relative (fixes #932).
   380	    """
   381	    if not p:
   382	        return p
   383	    p = p.replace("\\", "/")
   384	    if root and _is_abs(p):
   385	        try:
   386	            p = Path(p).relative_to(root).as_posix()
   387	        except ValueError:
   388	            # Lexical relative_to failed. Retry with both sides fully resolved:
